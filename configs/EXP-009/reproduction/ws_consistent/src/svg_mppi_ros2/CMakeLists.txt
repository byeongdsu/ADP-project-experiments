cmake_minimum_required(VERSION 3.16)
project(svg_mppi_ros2)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(grid_map_core REQUIRED)
find_package(grid_map_ros REQUIRED)
find_package(grid_map_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(svg_mppi_core src/mpc_base.cpp src/prior_samples_with_costs.cpp src/forward_mppi.cpp src/stein_variational_guided_mppi.cpp)
target_include_directories(svg_mppi_core PUBLIC include)
target_link_libraries(svg_mppi_core Eigen3::Eigen OpenMP::OpenMP_CXX)
ament_target_dependencies(svg_mppi_core grid_map_core)
target_compile_options(svg_mppi_core PRIVATE -O3)
add_executable(mppi_sim_node src/mppi_sim_node.cpp)
target_link_libraries(mppi_sim_node svg_mppi_core yaml-cpp)
ament_target_dependencies(mppi_sim_node rclcpp grid_map_ros grid_map_msgs ackermann_msgs nav_msgs sensor_msgs std_msgs)
install(TARGETS svg_mppi_core mppi_sim_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/sim_inputs.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
